## सेटअप

```typescript
import Phaseo from "@phaseo/sdk";

const client = new Phaseo({ apiKey: process.env.PHASEO_API_KEY! });
```

## चैट कम्प्लीशन

```typescript
const completion = await client.generateText({
  model: "openai/gpt-4o-mini",
  messages: [
    { role: "system", content: "You are a helpful assistant." },
    { role: "user", content: "What is Phaseo?" },
  ],
  temperature: 0.7,
});

for await (const chunk of client.streamChat({
  model: "openai/gpt-4o-mini",
  messages: [{ role: "user", content: "Tell me a story" }],
})) {
  if (chunk.text) process.stdout.write(chunk.text);
}
```

## Responses

```typescript
const response = await client.generateResponse({
  model: "openai/gpt-4.1",
  input: [{ role: "user", content: [{ type: "input_text", text: "Summarise this" }] }],
});

for await (const chunk of client.streamResponses({
  model: "openai/gpt-4.1",
  input: [{ role: "user", content: [{ type: "input_text", text: "Stream this" }] }],
  stream: true,
})) {
  if (chunk.text) process.stdout.write(chunk.text);
}
```

## बैच और फ़ाइलें

```typescript
const file = await client.uploadFile({ purpose: "batch", file: new Blob(["{}"], { type: "application/json" }) });
const batch = await client.createBatch({
  input_file_id: file.id,
  endpoint: "responses",
  completion_window: "24h",
});
const batchStatus = await client.getBatch(batch.id);
const files = await client.listFiles();
```
